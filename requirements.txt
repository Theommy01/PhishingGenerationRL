# Full install: dashboard, training/generation, the SVM detector and the tests.
#
#   pip install -r requirements.txt
#
# The dependency sets live in pyproject.toml as optional-dependency extras, so
# there is one list to maintain rather than several drifting copies:
#
#   pip install '.[dashboard]'   dashboard + data backend, no torch/unsloth
#   pip install '.[train]'       generation and the KTO/BCO trainers (heavy)
#   pip install '.[detectors]'   the SVM baseline detector
#   pip install '.[dev]'         pytest
#
# unsloth pins a specific torch/CUDA build; install the training extra into an
# environment that already has a working torch. MongoDB must be running on
# localhost:27017 for the store (the tests skip rather than fail without it).
.[all]
